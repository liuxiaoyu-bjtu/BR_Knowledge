
===== robot_info =====
signature: (self) -> 'RobotInfo'
doc: Return robot metadata.

Returns:
    RobotInfo: Robot name, model, serial number, and related metadata.
    @property
    @override
    def robot_info(self) -> RobotInfo:
        """Return robot metadata.

        Returns:
            RobotInfo: Robot name, model, serial number, and related metadata.
        """
        info = self.loco_client.GetRobotInfo()
        cli_info_getter = getattr(self.loco_client, "_get_cli_robot_info", None)
        cli_info = cli_info_getter() if callable(cli_info_getter) else {}
        extra = {
            key: value
            for key, value in cli_info.items()
            if key not in _ROBOT_INFO_FIELD_NAMES
        }

        return RobotInfo(
            name=info.name or "",
            model=info.model or "",
            serial_number=info.serial_number or "",
            firmware_version=info.version or "",
            manufacturer="Booster Robotics",
            extra=extra,
        )


===== get_mode =====
signature: (self) -> 'RobotModeName'
doc: Return the current robot mode.

Returns:
    str: Robot mode. Possible values include:
        - "damping": Safe damping mode, with highest priority.
        - "prepare": Preparation mode required before walking.
        - "walk": Walking mode for base velocity commands.
        - "custom": Custom mode for user-provided joint commands.
    def get_mode(self) -> RobotModeName:
        """Return the current robot mode.

        Returns:
            str: Robot mode. Possible values include:
                - "damping": Safe damping mode, with highest priority.
                - "prepare": Preparation mode required before walking.
                - "walk": Walking mode for base velocity commands.
                - "custom": Custom mode for user-provided joint commands.
        """
        try:
            return self._public_robot_mode_name(self._get_raw_mode())
        except Exception:
            logger.error("Failed to get robot mode")
            raise RuntimeError("Failed to get robot mode")

===== list_sensors =====
signature: (self) -> 'list[SensorInfo]'
doc: Return sensor metadata for all mounted sensors.

Booster robots expose IMU and camera information via separate SDK RPCs.
Virtual robots return mock sensor data to avoid SDK dependency.

Returns:
    list[SensorInfo]: List of all sensor metadata.

Raises:
    RuntimeError: Querying the sensor list from the SDK failed
        (real robot only).
    @override
    def list_sensors(self) -> list[SensorInfo]:
        """Return sensor metadata for all mounted sensors.

        Booster robots expose IMU and camera information via separate SDK RPCs.
        Virtual robots return mock sensor data to avoid SDK dependency.

        Returns:
            list[SensorInfo]: List of all sensor metadata.

        Raises:
            RuntimeError: Querying the sensor list from the SDK failed
                (real robot only).
        """
        if self._is_virtual_robot():
            return get_mock_sensors_for_virtual_robot()

        try:
            return fetch_all_sensors(self.loco_client, self.camera_client)
        except Exception:
            logger.error("Failed to list sensors")
            raise RuntimeError("Failed to list sensors")
